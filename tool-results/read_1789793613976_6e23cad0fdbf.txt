    13→    13→    • On page load the card DROPS from the top, the strap snaps taut and it swings.

   229→   229→<script>
   230→   230→(() => {
   231→   231→'use strict';
   232→   232→
   233→   233→/* =========================================================================
   234→   234→   CONFIG — edit here (or call HallPass.configure({...}) at runtime)
   235→   235→   ========================================================================= */
   236→   236→const CONFIG = {
   237→   237→  eyebrow : 'Staff invite',
   238→   238→  number  : '0042',
   239→   239→  title   : 'Hall Pass',
   240→   240→  subtitle: 'Scan to join — role and school are filled in for you.',
   241→   241→  path    : '/register/teacher',
   242→   242→  token   : '7F3-91C',
   243→   243→  usage   : 'one-time use',
   244→   244→  perks   : ['Role auto-assigned', 'School auto-linked', 'Expires after first scan'],
   245→   245→  school  : 'Your school',
   246→   246→  qr      : null            // SVG string | image URL | DOM element. null = placeholder pattern
   247→   247→};
   248→   248→
   249→   249→/* SIM:START ============================================================
   250→   250→   Physics. DOM-free on purpose so it can be unit-tested / reused.
   251→   251→   Coordinates are px, origin = top-left of the stage.
   252→   252→   ======================================================================= */
   253→   253→const TAU = Math.PI * 2;
   254→   254→const clamp = (v, a, b) => Math.max(a, Math.min(b, v));
   255→   255→const DT = 1 / 120;
   256→   256→
   257→   257→function createSim(o) {
   258→   258→  const { ax, ay, W, H, L, gap, width, height, drop } = o;
   259→   259→  const G = 4200, DAMP = 0.9965, ITERS = 10, N = 9, seg = L / N;
   260→   260→
   261→   261→  const pts = [];
   262→   262→  const mk = (x, y, im) => { const p = { x, y, px: x, py: y, im }; pts.push(p); return p; };
   263→   263→
   264→   264→  /* --- initial pose --------------------------------------------------- */
   265→   265→  const th0   = drop ? 0.32 : 0;
   266→   266→  const cx0   = drop ? ax + Math.min(46, L * 0.15) : ax;
   267→   267→  const cy0   = drop ? ay - L * 0.92 : ay + L;      // dropping: rope folded up, card above → falls
   268→   268→  const anchor = mk(ax, ay, 0);
   269→   269→  const rope = [anchor];
   270→   270→  for (let i = 1; i < N; i++) {
   271→   271→    const t = i / N;
   272→   272→    rope.push(mk(ax + (cx0 - ax) * t, ay + (cy0 - ay) * t, 2.2));      // light rope points
   273→   273→  }
   274→   274→  const clip = mk(cx0, cy0, 1); rope.push(clip);
   275→   275→
   276→   276→  const cs = Math.cos(th0), sn = Math.sin(th0);
   277→   277→  const place = (lx, ly) => mk(cx0 + lx * cs - ly * sn, cy0 + lx * sn + ly * cs, 1);
   278→   278→  const TL = place(-W / 2, gap), TR = place(W / 2, gap), BR = place(W / 2, gap + H), BL = place(-W / 2, gap + H);
   279→   279→  const cardPts = [clip, TL, TR, BR, BL];
   280→   280→  const local = [[0, 0], [-W / 2, gap], [W / 2, gap], [W / 2, gap + H], [-W / 2, gap + H]];
   281→   281→
   282→   282→  /* --- constraints ---------------------------------------------------- */
   283→   283→  const cons = [];
   284→   284→  for (let i = 0; i < N; i++) cons.push({ a: rope[i], b: rope[i + 1], rest: seg, max: true });        // rope: can go slack, can’t stretch
   285→   285→  for (let i = 0; i < 5; i++) for (let j = i + 1; j < 5; j++) {                                      // card: rigid
   286→   286→    cons.push({ a: cardPts[i], b: cardPts[j], rest: Math.hypot(local[i][0] - local[j][0], local[i][1] - local[j][1]), max: false });
   287→   287→  }
   288→   288→
   289→   289→  if (drop) for (const p of pts) if (p.im) p.py = p.y - 700 * DT;    // initial downward speed
   290→   290→
   291→   291→  const sim = {
   292→   292→    pts, rope, clip, TL, TR, BR, BL, W, H,
   293→   293→    drag: null,
   294→   294→    flipped: false,
   295→   295→    yaw: 0, yawV: 0, pitch: 0, pitchV: 0,
   296→   296→    hoverYaw: 0, hoverPitch: 0,
   297→   297→    gx: 0.5, gy: 0.2,
   298→   298→    time: 0,
   299→   299→
   300→   300→    step() {
   301→   301→      const g2 = G * DT * DT;
   302→   302→      for (const p of pts) if (p.im) {
   303→   303→        const vx = (p.x - p.px) * DAMP, vy = (p.y - p.py) * DAMP;
   304→   304→        p.px = p.x; p.py = p.y; p.x += vx; p.y += vy + g2;
   305→   305→      }
   306→   306→      if (this.drag) this._applyDrag();
   307→   307→      for (let it = 0; it < ITERS; it++) for (const c of cons) {
   308→   308→        const a = c.a, b = c.b;
   309→   309→        let dx = b.x - a.x, dy = b.y - a.y;
   310→   310→        const d = Math.hypot(dx, dy) || 1e-6;
   311→   311→        if (c.max && d <= c.rest) continue;
   312→   312→        const w = a.im + b.im; if (!w) continue;
   313→   313→        const diff = (d - c.rest) / d;
   314→   314→        a.x += dx * diff * (a.im / w); a.y += dy * diff * (a.im / w);
   315→   315→        b.x -= dx * diff * (b.im / w); b.y -= dy * diff * (b.im / w);
   316→   316→      }
   317→   317→      const pad = 4;
   318→   318→      for (const p of pts) if (p.im) {
   319→   319→        if (p.x < pad) p.x = pad; else if (p.x > width - pad) p.x = width - pad;
   320→   320→        if (p.y > height - pad) { p.y = height - pad; p.py = p.y + (p.py - p.y) * 0.2; }
   321→   321→      }
   322→   322→
   323→   323→      /* --- 3D: yaw / pitch as damped springs driven by the card’s motion --- */
   324→   324→      let vx = 0, vy = 0;
   325→   325→      for (const p of [TL, TR, BR, BL]) { vx += p.x - p.px; vy += p.y - p.py; }
   326→   326→      vx = vx / 4 / DT; vy = vy / 4 / DT;
   327→   327→      const base = this.flipped ? Math.PI : 0;
   328→   328→      const k = Math.round((this.yaw - base) / TAU);
   329→   329→      const yTarget = base + k * TAU + this.hoverYaw;
   330→   330→      const yAcc = -46 * (this.yaw - yTarget) - 6.2 * this.yawV + clamp(0.045 * vx, -40, 40);
   331→   331→      this.yawV = clamp(this.yawV + yAcc * DT, -32, 32); this.yaw += this.yawV * DT;
   332→   332→
   333→   333→      const pTarget = this.hoverPitch + clamp(-vy * 0.00025, -0.35, 0.35);
   334→   334→      const pAcc = -60 * (this.pitch - pTarget) - 9 * this.pitchV;
   335→   335→      this.pitchV += pAcc * DT; this.pitch += this.pitchV * DT;
   336→   336→      this.time += DT;
   337→   337→    },
   338→   338→
   339→   339→    _applyDrag() {
   340→   340→      const d = this.drag, P = [TL, TR, BR, BL];
   341→   341→      const w = [(1 - d.u) * (1 - d.v), d.u * (1 - d.v), d.u * d.v, (1 - d.u) * d.v];
   342→   342→      let gx = 0, gy = 0, s = 0;
   343→   343→      for (let i = 0; i < 4; i++) { gx += w[i] * P[i].x; gy += w[i] * P[i].y; s += w[i] * w[i]; }
   344→   344→      const k = 0.5, ex = (d.tx - gx) * k, ey = (d.ty - gy) * k;
   345→   345→      for (let i = 0; i < 4; i++) { P[i].x += (w[i] / s) * ex; P[i].y += (w[i] / s) * ey; }
   346→   346→    },
   347→   347→
   348→   348→    /* world point → card-local (0..1, 0..1) */
   349→   349→    toUV(x, y) {
   350→   350→      const ex = TR.x - TL.x, ey = TR.y - TL.y, fx = BL.x - TL.x, fy = BL.y - TL.y;
   351→   351→      const px = x - TL.x, py = y - TL.y;
   352→   352→      return { u: (px * ex + py * ey) / (W * W), v: (px * fx + py * fy) / (H * H) };
   353→   353→    },
   354→   354→    angle() { return Math.atan2(TR.y - TL.y, TR.x - TL.x); },
   355→   355→
   356→   356→    nudge(vx, vy) {
   357→   357→      const f = DT * 0.02;
   358→   358→      for (const p of [TL, TR, BR, BL, clip]) { p.px -= vx * f; p.py -= vy * f; }
   359→   359→      this.yawV += clamp(vx * 0.0012, -6, 6);
   360→   360→    },
   361→   361→    flip() { this.flipped = !this.flipped; this.yawV += this.flipped ? 7 : -7; },
   362→   362→    settle(seconds) { for (let i = 0; i < seconds / DT; i++) this.step(); }
   363→   363→  };
   364→   364→  return sim;
   365→   365→}

   480→   480→function placeholderQR(seed) {
   481→   481→  const n = 25;
   482→   482→  let h = 2166136261 >>> 0;
   483→   483→  for (const ch of String(seed)) { h ^= ch.charCodeAt(0); h = Math.imul(h, 16777619) >>> 0; }
   484→   484→  const rnd = () => { h = (h + 0x6D2B79F5) >>> 0; let t = Math.imul(h ^ (h >>> 15), 1 | h); t ^= t + Math.imul(t ^ (t >>> 7), 61 | t); return ((t ^ (t >>> 14)) >>> 0) / 4294967296; };
   485→   485→  const finder = (i, j) => (i === 0 || i === 6 || j === 0 || j === 6 || (i >= 2 && i <= 4 && j >= 2 && j <= 4));
   486→   486→  let rects = '';
   487→   487→  for (let r = 0; r < n; r++) for (let c = 0; c < n; c++) {
   488→   488→    let on;
   489→   489→    const inTL = r < 8 && c < 8, inTR = r < 8 && c >= n - 8, inBL = r >= n - 8 && c < 8;
   490→   490→    if (inTL || inTR || inBL) {
   491→   491→      const i = inBL ? r - (n - 8) : r, j = inTR ? c - (n - 8) : c;
   492→   492→      on = i < 7 && j < 7 && finder(i, j);
   493→   493→    } else on = rnd() > 0.52;
   494→   494→    if (on) rects += `<rect x="${c}" y="${r}" width="1.02" height="1.02"/>`;
   495→   495→  }
   496→   496→  return `<svg viewBox="0 0 ${n} ${n}" shape-rendering="crispEdges" fill="currentColor" role="img" aria-label="QR code placeholder">${rects}</svg>`;
   497→   497→}
