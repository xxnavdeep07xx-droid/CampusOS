     9→  ============================================================================
    10→  HALL PASS — 3D LANYARD CARD (single file, zero dependencies)
    11→  ============================================================================
    12→  WHAT IT DOES
    13→    • On page load the card DROPS from the top, the strap snaps taut and it swings.
    14→    • Real 2D verlet physics (rope + rigid card) with a 3D layer on top
    15→      (yaw / pitch / thickness / glare), rendered with CSS 3D transforms.
    16→    • Mouse / touch:  drag it, fling it, brush past it to nudge it,
    17→      hover to tilt it toward the cursor, click/tap (no drag) to flip it.
    18→
    19→  HOW TO INTEGRATE (notes for the coding agent)
    20→    1. Everything lives in three blocks: <style>, the #hpStage markup, and the
    21→       <script>. Copy them into the register/invite page as-is.
    22→    2. The stage fills its parent. Give #hpStage a height (default: fills the
    23→       viewport minus the demo header). The strap enters from the TOP EDGE of the
    24→       stage, so place the stage directly under your site header.
    25→    3. Edit the CONFIG object at the top of the <script> — or call
    26→         HallPass.configure({ number:'0043', path:'/register/teacher', token:'A1B-22C' })
    27→       at runtime, e.g. from server-rendered invite data.
    28→    4. THE QR CODE IS A PLACEHOLDER (not scannable). Replace it with a real one:
    29→         HallPass.setQR('<svg …>…</svg>')   // SVG markup string
    30→         HallPass.setQR('https://…/qr.png') // image URL / data: URI
    31→         HallPass.setQR(domElement)         // any element (canvas, img, svg)
    32→       Generate the QR from the full invite URL (path + ?token=…), e.g. with the
    33→       `qrcode` npm package on the server or client.
    34→    5. Other API:  HallPass.drop()  → replay the fall,  HallPass.flip() → flip card.
    35→    6. Colors / fonts are CSS variables in :root (--hp-*). The card scales with

   229→<script>
   230→(() => {
   231→'use strict';
   232→
   233→/* =========================================================================
   234→   CONFIG — edit here (or call HallPass.configure({...}) at runtime)
   235→   ========================================================================= */
   236→const CONFIG = {
   237→  eyebrow : 'Staff invite',
   238→  number  : '0042',
   239→  title   : 'Hall Pass',
   240→  subtitle: 'Scan to join — role and school are filled in for you.',
   241→  path    : '/register/teacher',
   242→  token   : '7F3-91C',
   243→  usage   : 'one-time use',
   244→  perks   : ['Role auto-assigned', 'School auto-linked', 'Expires after first scan'],
   245→  school  : 'Your school',
   246→  qr      : null            // SVG string | image URL | DOM element. null = placeholder pattern
   247→};
   248→
   249→/* SIM:START ============================================================
   250→   Physics. DOM-free on purpose so it can be unit-tested / reused.
   251→   Coordinates are px, origin = top-left of the stage.
   252→   ======================================================================= */
   253→const TAU = Math.PI * 2;
   254→const clamp = (v, a, b) => Math.max(a, Math.min(b, v));
   255→const DT = 1 / 120;
   256→
   257→function createSim(o) {
   258→  const { ax, ay, W, H, L, gap, width, height, drop } = o;
   259→  const G = 4200, DAMP = 0.9965, ITERS = 10, N = 9, seg = L / N;
   260→
   261→  const pts = [];
   262→  const mk = (x, y, im) => { const p = { x, y, px: x, py: y, im }; pts.push(p); return p; };
   263→
   264→  /* --- initial pose --------------------------------------------------- */
   265→  const th0   = drop ? 0.32 : 0;
   266→  const cx0   = drop ? ax + Math.min(46, L * 0.15) : ax;
   267→  const cy0   = drop ? ay - L * 0.92 : ay + L;      // dropping: rope folded up, card above → falls
   268→  const anchor = mk(ax, ay, 0);
   269→  const rope = [anchor];
   270→  for (let i = 1; i < N; i++) {
   271→    const t = i / N;
   272→    rope.push(mk(ax + (cx0 - ax) * t, ay + (cy0 - ay) * t, 2.2));      // light rope points
   273→  }
   274→  const clip = mk(cx0, cy0, 1); rope.push(clip);
   275→
   276→  const cs = Math.cos(th0), sn = Math.sin(th0);
   277→  const place = (lx, ly) => mk(cx0 + lx * cs - ly * sn, cy0 + lx * sn + ly * cs, 1);
   278→  const TL = place(-W / 2, gap), TR = place(W / 2, gap), BR = place(W / 2, gap + H), BL = place(-W / 2, gap + H);
   279→  const cardPts = [clip, TL, TR, BR, BL];
   280→  const local = [[0, 0], [-W / 2, gap], [W / 2, gap], [W / 2, gap + H], [-W / 2, gap + H]];
   281→
   282→  /* --- constraints ---------------------------------------------------- */
   283→  const cons = [];
   284→  for (let i = 0; i < N; i++) cons.push({ a: rope[i], b: rope[i + 1], rest: seg, max: true });        // rope: can go slack, can’t stretch
   285→  for (let i = 0; i < 5; i++) for (let j = i + 1; j < 5; j++) {                                      // card: rigid
   286→    cons.push({ a: cardPts[i], b: cardPts[j], rest: Math.hypot(local[i][0] - local[j][0], local[i][1] - local[j][1]), max: false });
   287→  }
   288→
   289→  if (drop) for (const p of pts) if (p.im) p.py = p.y - 700 * DT;    // initial downward speed
   290→
   291→  const sim = {
   292→    pts, rope, clip, TL, TR, BR, BL, W, H,
   293→    drag: null,
   294→    flipped: false,
   295→    yaw: 0, yawV: 0, pitch: 0, pitchV: 0,
   296→    hoverYaw: 0, hoverPitch: 0,
   297→    gx: 0.5, gy: 0.2,
   298→    time: 0,
   299→
   300→    step() {
   301→      const g2 = G * DT * DT;
   302→      for (const p of pts) if (p.im) {
   303→        const vx = (p.x - p.px) * DAMP, vy = (p.y - p.py) * DAMP;
   304→        p.px = p.x; p.py = p.y; p.x += vx; p.y += vy + g2;
   305→      }
   306→      if (this.drag) this._applyDrag();
   307→      for (let it = 0; it < ITERS; it++) for (const c of cons) {
   308→        const a = c.a, b = c.b;
   309→        let dx = b.x - a.x, dy = b.y - a.y;
   310→        const d = Math.hypot(dx, dy) || 1e-6;
   311→        if (c.max && d <= c.rest) continue;
   312→        const w = a.im + b.im; if (!w) continue;
   313→        const diff = (d - c.rest) / d;
   314→        a.x += dx * diff * (a.im / w); a.y += dy * diff * (a.im / w);
   315→        b.x -= dx * diff * (b.im / w); b.y -= dy * diff * (b.im / w);
   316→      }
   317→      const pad = 4;
   318→      for (const p of pts) if (p.im) {
   319→        if (p.x < pad) p.x = pad; else if (p.x > width - pad) p.x = width - pad;
   320→        if (p.y > height - pad) { p.y = height - pad; p.py = p.y + (p.py - p.y) * 0.2; }
   321→      }
   322→
   323→      /* --- 3D: yaw / pitch as damped springs driven by the card’s motion --- */
   324→      let vx = 0, vy = 0;
   325→      for (const p of [TL, TR, BR, BL]) { vx += p.x - p.px; vy += p.y - p.py; }
   326→      vx = vx / 4 / DT; vy = vy / 4 / DT;
   327→      const base = this.flipped ? Math.PI : 0;
   328→      const k = Math.round((this.yaw - base) / TAU);
   329→      const yTarget = base + k * TAU + this.hoverYaw;
   330→      const yAcc = -46 * (this.yaw - yTarget) - 6.2 * this.yawV + clamp(0.045 * vx, -40, 40);
   331→      this.yawV = clamp(this.yawV + yAcc * DT, -32, 32); this.yaw += this.yawV * DT;
   332→
   333→      const pTarget = this.hoverPitch + clamp(-vy * 0.00025, -0.35, 0.35);
   334→      const pAcc = -60 * (this.pitch - pTarget) - 9 * this.pitchV;
   335→      this.pitchV += pAcc * DT; this.pitch += this.pitchV * DT;
   336→      this.time += DT;
   337→    },
   338→
   339→    _applyDrag() {
   340→      const d = this.drag, P = [TL, TR, BR, BL];
   341→      const w = [(1 - d.u) * (1 - d.v), d.u * (1 - d.v), d.u * d.v, (1 - d.u) * d.v];
   342→      let gx = 0, gy = 0, s = 0;
   343→      for (let i = 0; i < 4; i++) { gx += w[i] * P[i].x; gy += w[i] * P[i].y; s += w[i] * w[i]; }
   344→      const k = 0.5, ex = (d.tx - gx) * k, ey = (d.ty - gy) * k;
   345→      for (let i = 0; i < 4; i++) { P[i].x += (w[i] / s) * ex; P[i].y += (w[i] / s) * ey; }
   346→    },
   347→
   348→    /* world point → card-local (0..1, 0..1) */
   349→    toUV(x, y) {
   350→      const ex = TR.x - TL.x, ey = TR.y - TL.y, fx = BL.x - TL.x, fy = BL.y - TL.y;
   351→      const px = x - TL.x, py = y - TL.y;
   352→      return { u: (px * ex + py * ey) / (W * W), v: (px * fx + py * fy) / (H * H) };
   353→    },
   354→    angle() { return Math.atan2(TR.y - TL.y, TR.x - TL.x); },
   355→
   356→    nudge(vx, vy) {
   357→      const f = DT * 0.02;
   358→      for (const p of [TL, TR, BR, BL, clip]) { p.px -= vx * f; p.py -= vy * f; }
   359→      this.yawV += clamp(vx * 0.0012, -6, 6);
   360→    },
   361→    flip() { this.flipped = !this.flipped; this.yawV += this.flipped ? 7 : -7; },
   362→    settle(seconds) { for (let i = 0; i < seconds / DT; i++) this.step(); }
   363→  };
   364→  return sim;
   365→}
   366→/* SIM:END ================================================================ */

   397→function strapPath(rope) {
   398→  const p = rope;
   399→  let d = `M${p[0].x.toFixed(1)},${(p[0].y - 120).toFixed(1)} L${p[0].x.toFixed(1)},${p[0].y.toFixed(1)}`;
   400→  for (let i = 0; i < p.length - 1; i++) {
   401→    const p0 = p[i - 1] || p[i], p1 = p[i], p2 = p[i + 1], p3 = p[i + 2] || p2;
   402→    d += ` C${(p1.x + (p2.x - p0.x) / 6).toFixed(1)},${(p1.y + (p2.y - p0.y) / 6).toFixed(1)} ` +
   403→         `${(p2.x - (p3.x - p1.x) / 6).toFixed(1)},${(p2.y - (p3.y - p1.y) / 6).toFixed(1)} ` +
   404→         `${p2.x.toFixed(1)},${p2.y.toFixed(1)}`;
   405→  }
   406→  return d;
   407→}

   479→/* ---- content + QR ---- */
   480→function placeholderQR(seed) {
   481→  const n = 25;
   482→  let h = 2166136261 >>> 0;
   483→  for (const ch of String(seed)) { h ^= ch.charCodeAt(0); h = Math.imul(h, 16777619) >>> 0; }
   484→  const rnd = () => { h = (h + 0x6D2B79F5) >>> 0; let t = Math.imul(h ^ (h >>> 15), 1 | h); t ^= t + Math.imul(t ^ (t >>> 7), 61 | t); return ((t ^ (t >>> 14)) >>> 0) / 4294967296; };
   485→  const finder = (i, j) => (i === 0 || i === 6 || j === 0 || j === 6 || (i >= 2 && i <= 4 && j >= 2 && j <= 4));
   486→  let rects = '';
   487→  for (let r = 0; r < n; r++) for (let c = 0; c < n; c++) {
   488→    let on;
   489→    const inTL = r < 8 && c < 8, inTR = r < 8 && c >= n - 8, inBL = r >= n - 8 && c < 8;
   490→    if (inTL || inTR || inBL) {
   491→      const i = inBL ? r - (n - 8) : r, j = inTR ? c - (n - 8) : c;
   492→      on = i < 7 && j < 7 && finder(i, j);
   493→    } else on = rnd() > 0.52;
   494→    if (on) rects += `<rect x="${c}" y="${r}" width="1.02" height="1.02"/>`;
   495→  }
   496→  return `<svg viewBox="0 0 ${n} ${n}" shape-rendering="crispEdges" fill="currentColor" role="img" aria-label="QR code placeholder">${rects}</svg>`;
   497→}
