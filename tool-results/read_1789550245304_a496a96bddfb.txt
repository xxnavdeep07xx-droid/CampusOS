   101→/**
   102→ * Build the public-facing URL that an invited user should visit.
   103→ *
   104→ * Routes:
   105→ *   - principal → no token, direct register flow (not used; principals self-register)
   106→ *   - staff / teacher → /register/teacher?token=...
   107→ *   - student → /register/student?token=...
   108→ *
   109→ * Note: staff and teachers share the same registration flow because staff
   110→ * members are effectively teachers without an assigned class.
   111→ */
   112→export function inviteUrl(token: string, role: UserRole): string {
   113→  const site = process.env.NEXT_PUBLIC_SITE_URL ?? "http://localhost:3000";
   114→  const sub =
   115→    role === "student" ? "student" : role === "teacher" ? "teacher" : "teacher";
   116→  return `${site}/register/${sub}?token=${token}`;
   117→}

   263→/**
   264→ * Map a file extension or MIME type to a brutalist accent color class.
   265→ *
   266→ * Used by ResourceCard + AssignmentCard to color-code by file type.
   267→ */
   268→export type FileColor =
   269→  | "sky"     // PDFs
   270→  | "emerald" // docs / text
   271→  | "coral"   // images
   272→  | "violet"  // video
   273→  | "amber"   // archives
   274→  | "slate";  // default
   275→
   276→export function fileColorFor(filename: string, mimeType?: string | null): FileColor {
   277→  const ext = filename.toLowerCase().split(".").pop() ?? "";
   278→  if (mimeType?.startsWith("image/") || ["png", "jpg", "jpeg", "gif", "webp", "svg"].includes(ext)) {
   279→    return "coral";
   280→  }
   281→  if (mimeType?.startsWith("video/") || ["mp4", "mov", "avi", "mkv", "webm"].includes(ext)) {
   282→    return "violet";
   283→  }
   284→  if (mimeType?.startsWith("audio/") || ["mp3", "wav", "ogg", "m4a"].includes(ext)) {
   285→    return "violet";
   286→  }
   287→  if (["zip", "rar", "7z", "tar", "gz"].includes(ext)) {
   288→    return "amber";
   289→  }
   290→  if (["pdf"].includes(ext)) {
   291→    return "sky";
   292→  }
   293→  if (["doc", "docx", "txt", "md", "rtf", "odt"].includes(ext)) {
   294→    return "emerald";
   295→  }
   296→  if (["xls", "xlsx", "csv", "ppt", "pptx"].includes(ext)) {
   297→    return "amber";
   298→  }
   299→  return "slate";
   300→}

   419→/**
   420→ * Parse "HH:MM:SS" → minutes-since-midnight. Returns 0 for null/invalid.
   421→ */
   422→export function timeToMinutes(t: string | null | undefined): number {
   423→  if (!t) return 0;
   424→  const m = /^(\d{1,2}):(\d{2})(?::(\d{2}))?$/.exec(t);
   425→  if (!m) return 0;
   426→  return parseInt(m[1], 10) * 60 + parseInt(m[2], 10);
   427→}
   428→
   429→/**
   430→ * Format "HH:MM:SS" → "h:mm AM/PM" (e.g. "14:30:00" → "2:30 PM").
   431→ */
   432→export function formatTime(t: string | null | undefined): string {
   433→  if (!t) return "—";
   434→  const mins = timeToMinutes(t);
   435→  const h24 = Math.floor(mins / 60);
   436→  const m = mins % 60;
   437→  const ampm = h24 >= 12 ? "PM" : "AM";
   438→  const h12 = h24 % 12 === 0 ? 12 : h24 % 12;
   439→  return `${h12}:${String(m).padStart(2, "0")} ${ampm}`;
   440→}
   441→
   442→/**
   443→ * Convert a Date to YYYY-MM-DD in local timezone (for the `date` column).
   444→ */
   445→export function toDateInputValue(d: Date): string {
   446→  const y = d.getFullYear();
   447→  const m = String(d.getMonth() + 1).padStart(2, "0");
   448→  const day = String(d.getDate()).padStart(2, "0");
   449→  return `${y}-${m}-${day}`;
   450→}

   607→/**
   608→ * Compute the score for a single question given the student's answer.
   609→ *
   610→ * - MCQ: case-insensitive exact match against one of the options.
   611→ * - short_answer: case-insensitive trimmed exact match against correct_answer.
   612→ *
   613→ * Returns the question's `points` if correct, 0 otherwise.
   614→ */
   615→export function gradeQuestion(
   616→  question: Pick<QuizQuestion, "question_type" | "correct_answer" | "points">,
   617→  studentAnswer: string | undefined
   618→): number {
   619→  if (!studentAnswer) return 0;
   620→  const ans = studentAnswer.trim().toLowerCase();
   621→  const correct = question.correct_answer.trim().toLowerCase();
   622→  if (!ans || !correct) return 0;
   623→  return ans === correct ? question.points : 0;
   624→}
   625→
   626→/**
   627→ * Compute total score + max score for a quiz attempt.
   628→ *
   629→ * Returns { score, maxScore }.
   630→ */
   631→export function gradeAttempt(
   632→  questions: QuizQuestion[],
   633→  answers: Record<string, string>
   634→): { score: number; maxScore: number } {
   635→  let score = 0;
   636→  let maxScore = 0;
   637→  for (const q of questions) {
   638→    maxScore += q.points;
   639→    score += gradeQuestion(q, answers[q.id]);
   640→  }
   641→  return { score, maxScore };
   642→}

   648→export function gradeColor(pct: number | null | undefined): string {
   649→  if (pct == null) return "bg-slate-300";
   650→  if (pct >= 90) return "bg-emerald-500";
   651→  if (pct >= 75) return "bg-amber-400";
   652→  if (pct >= 50) return "bg-sky-300";
   653→  return "bg-rose-400";

   739→/**
   740→ * Format a decimal amount as a currency string.
   741→ * Defaults to USD with no symbol-prefix customization — adjust as needed.
   742→ */
   743→export function formatCurrency(amount: number | string | null | undefined): string {
   744→  if (amount == null) return "$0.00";
   745→  const n = typeof amount === "string" ? parseFloat(amount) : amount;
   746→  if (Number.isNaN(n)) return "$0.00";
   747→  return new Intl.NumberFormat("en-US", {
   748→    style: "currency",
   749→    currency: "USD",
   750→    minimumFractionDigits: 2,
   751→    maximumFractionDigits: 2,
   752→  }).format(n);
   753→}
